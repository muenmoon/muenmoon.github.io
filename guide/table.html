# 表格数据

TIP

通过绑定 loadData 事件获取数据后传递给属性 tableData 完成表格数据渲染；同时数据加载过程状态可以配置属性 loading 实现。

人员列表

人员列表

暂无数据
暂无数据
加载中...
共 0 条记录1
<template>
  <BuseCrud
    ref="crud"
    title="人员列表"
    :loading="loading"
    :tablePage="tablePage"
    :tableColumn="tableColumn"
    :tableData="tableData"
    :modalConfig="modalConfig"
    @loadData="loadData"
  >
  </BuseCrud>
</template>

<script>
  const res = {
    data: [
      { name: "张三", sex: "1", age: 18 },
      { name: "李四", sex: "2", age: 19 },
    ],
    total: 2,
  };
  export default {
    name: "DumiDocVueIndex",
    data() {
      return {
        tableData: [],
        tableColumn: [
          {
            field: "name",
            title: "姓名",
          },
          {
            field: "sex",
            title: "性别",
            formatter: ({ cellValue }) => {
              return cellValue === "1" ? "男" : "女";
            },
          },
          {
            field: "age",
            title: "年龄",
          },
        ],
        params: { name: "", sex: "", age: "" },
        loading: false,
        tablePage: { total: 0, currentPage: 1, pageSize: 10 },
      };
    },
    computed: {
      modalConfig() {
        return {
          addBtn: false,
          menu: false,
        };
      },
    },
    mounted() {
      this.loadData();
    },
    methods: {
      async loadData() {
        this.loading = true;
        const result = await new Promise((resolve) => {
          setTimeout(() => {
            resolve(res);
          }, 1000);
        });
        this.loading = false;
        this.tableData = result.data;
        this.tablePage.total = result.total;
      },
    },
  };
</script>
