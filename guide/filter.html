点击查询按钮后，会调用 loadData 方法。

<template>
  <BuseCrud
    ref="crud"
    title="人员列表"
    :loading="loading"
    :filterOptions="filterOptions"
    :tablePage="tablePage"
    :tableColumn="tableColumn"
    :tableData="tableData"
    :modalConfig="modalConfig"
    @modalCancel="modalCancelHandler"
    @modalSubmit="modalSubmit"
    @modalConfirm="modalConfirmHandler"
    @rowDel="deleteRowHandler"
    @rowEdit="rowEditHandler"
    @loadData="loadData"
  >
  </BuseCrud>
</template>

<script>
  const res = {
    data: [
      { name: "张三", sex: "1", age: 18 },
      { name: "李四", sex: "2", age: 19 },
    ],
    total: 2,
  };
  export default {
    name: "DumiDocVueIndex",
    data() {
      return {
        menuShow: true,
        tableData: [],
        tableColumn: [
          {
            field: "name",
            title: "姓名",
          },
          {
            field: "sex",
            title: "性别",
            formatter: ({ cellValue }) => {
              return cellValue === "1" ? "男" : "女";
            },
          },
          {
            field: "age",
            title: "年龄",
          },
        ],
        params: { name: "", sex: "1", age: "" },
        loading: false,
        tablePage: { total: 0, currentPage: 1, pageSize: 10 },
      };
    },
    computed: {
      filterOptions() {
        return {
          config: [
            {
              field: "name",
              title: "姓名",
            },
            {
              field: "sex",
              title: "性别",
              element: "el-select",
              props: {
                options: [
                  {
                    label: "男",
                    value: "1",
                  },
                  {
                    label: "女",
                    value: "2",
                  },
                ],
              },
            },
            {
              field: "age",
              title: "年龄",
            },
          ],
          params: this.params,
        };
      },
      modalConfig() {
        return {
          addBtn: false,
          menu: false,
        };
      },
    },
    mounted() {},
    methods: {
      async loadData() {
        this.$message.info(`查询参数为：${JSON.stringify(this.params)}`);
        this.loading = true;
        const result = await new Promise((resolve) => {
          setTimeout(() => {
            //模拟数据
            const finalData = res.data.map((item) => {
              return {
                sex: this.params.sex || Math.round(Math.random()) + 1 + "",
                name: this.params.name || "zyj" + Math.ceil(Math.random() * 50),
                age: this.params.age || Math.ceil(Math.random() * 60),
              };
            });
            resolve({ ...res, data: finalData });
          }, 1000);
        });
        this.loading = false;
        this.tableData = result.data;
        this.tablePage.total = result.total;
      },
      modalCancelHandler() {},
      modalSubmit() {},
      modalCancelHandler() {},
      deleteRowHandler() {},
      modalConfirmHandler() {},
      rowEditHandler() {},
    },
  };
</script>

    total: 2,
  };
  export default {
    name: "DumiDocVueIndex",
    data() {
      return {
        menuShow: true,
        persons: null,
        tableData: [],
        //是否在请求人员数据
        fetching: false,
        sexOptions: [],
        tableColumn: [
          {
            field: "name",
            title: "姓名",
          },
          {
            field: "sex",
            title: "性别",
            formatter: ({ cellValue }) => {
              return cellValue === "1" ? "男" : "女";
            },
          },
          {
            field: "age",
            title: "年龄",
          },
          {
            field: "birth",
            title: "出生年",

        loading: false,
        tablePage: { total: 0, currentPage: 1, pageSize: 10 },
      };
    },
    computed: {
      filterOptions() {
        return {
          config: [
            {
              field: "name",
              title: "姓名",
              element: "el-select",
              props: {
                filterable: true,
                options: this.persons,
                placeholder: "请选择人员",
                loading: this.fetching,
              },
              on: {
                focus: this.getPersons,
              },
            },
            {
              field: "sex",
              title: "性别",
              element: "el-select",
              props: {
                options: this.sexOptions,
              },
            },
            {
              field: "age",
              title: "年龄",
            },
          ],
          params: this.params,
        };
      },
      modalConfig() {
        return {
          addBtn: false,
          menu: false,
        };
      },
    },
    mounted() {
      //获取性别数据
      this.getSex();
    },
    methods: {
      async getSex() {
        const res = await new Promise((resolve) => {
          setTimeout(() => {
            resolve([
              {
                label: "男",
                value: "1",
              },
              {
                label: "女",
                value: "2",
              },
            ]);
          }, 1000);
        });
        this.sexOptions = res;
      },
      async getPersons() {
        if (this.persons) return;
        this.fetching = true;
        const res = await new Promise((resolve) => {
          setTimeout(() => {
            resolve([
              {
                label: "小明",
                value: "1",
              },
              {
                label: "小红",
                value: "2",
              },
              {
                label: "小王",
                value: "3",
              },
            ]);
          }, 1000);
        });
        this.fetching = false;
        this.persons = res;
      },
      async loadData() {
        this.$message.info(`查询参数为：${JSON.stringify(this.params)}`);
        this.loading = true;
        const result = await new Promise((resolve) => {
          setTimeout(() => {
            //模拟数据
            const finalData = res.data.map((item) => {
              return {
                sex: this.params.sex || Math.round(Math.random()) + 1 + "",
                name: this.params.name || "zyj" + Math.ceil(Math.random() * 50),
                age: this.params.age || Math.ceil(Math.random() * 60),
